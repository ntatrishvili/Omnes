    # Ohm/km

Device:
Generator:

    p_out: data/pv1.csv in_column p
    q_out: data/pv1.csv in_column q
    tags:
      # Validate this: vector can only be electricity
      # Default vector: electricity; possible values of vector: electricity, heat, material,
      # Default contributes_to: electric_power_balance
      household: HH1
    relations:
      $.p_out <= $.peak_power
      $.p_out >= 0
      $.q_out <= 0.3 * $.p_out

  - pv2 on bus_LV2
    peak_power: 3
    # pv power input is in data/input.csv in column "pv2"
    p_out: data/pv2.csv in_column p_out

    tags:
      # Validate this: can only be electricity

  Wind:
  - id: wind1 on bus_LV3
    peak_power: 5

    # wind power input is in data/input.csv in column "wind"
    p_out: data/wind.csv in_column p_out
    q_out: data/wind.csv in_column q_out
    tags:
      # Default vector: electricity
      # Default contributes_to: electric_power_balance
      # No household here, does not belong to any specific household
    relations:
      $.p_out <= $.peak_power
      $.p_out >= 0
      $.p_out * $.efficiency <= $.peak_power
      $.q_out <= 0.2 * $.p_out

Storage:
 Battery:
  - battery1 on bus_LV3
    capacity: 5
    max_charge_rate: 2
    # default: max_charge_rate
    max_discharge_rate: 2
    # default: 1
    charge_efficiency: 0.95
    # default: 1
    discharge_efficiency: 0.95
    # default: 1
    storage_efficiency: 0.995
    p_in: input/battery1.csv in_column p_in
    p_out: input/battery1.csv in_column p_out
    tags:
        # Default vector: electricity
        # Default contributes_to: electric_power_balance
    relations:
      $.p_in <= $.max_charge_rate
      $.p_out <= $.max_discharge_rate
      $.state_of_charge = $.state_of_charge(t-1) * $.storage_efficiency + $.p_in * $.charge_efficiency - $.p_out / $.discharge_efficiency
      $.state_of_charge >= 0.1 * $.capacity
      $.state_of_charge <= $.capacity

    relations:

  - heater2 on bus_LV2 charges hot_water2
    # this is not controllable, so 'controllable' is not specified
    tags:
      is same as the input
      # Default vector: electricity
      # Default contributes_to: electric_power_balance
      household: HH2

      $.power <= 3

Slack:

Relations:

  # perfect = read from the data, persistence = value of the previous day
  forecast: perfect

  Episode:

    # random_day | sequential | full_period
    mode: random_day
